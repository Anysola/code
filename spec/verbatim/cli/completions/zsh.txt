###-begin-anysola-completions-###
#
# Static completion script for Zsh
#
# Installation:
#   anysola --completions zsh > ~/.zsh/completions/_anysola
#   then add ~/.zsh/completions to your fpath
#

_anysola() {
  local context state state_descr line
  typeset -A opt_args

  local -a commands
  commands=(
    'login:Sign in with your Anysola account'
    'logout:Revoke the Anysola token and remove it from this machine'
    'status:Show your Anysola plan and credits'
    'usage:Show Anysola usage\: totals, by day and by model'
    'devices:List the devices signed in to your Anysola account'
    'update:Update Anysola Code the way it was installed (install script or npm)'
    'upgrade:Update Anysola Code the way it was installed (install script or npm)'
    'uninstall:Uninstall Anysola Code, keeping session data, configuration, and state'
    'acp:Start an Agent Client Protocol server'
    'debug:Debugging and troubleshooting tools'
    'mcp:Manage MCP (Model Context Protocol) servers'
    'plugin:Manage plugins'
    'models:List all available models'
    'run:Run Anysola Code with a message'
    'session:Manage sessions'
    'service:Manage the background server'
    'reload:Reload configuration'
    'serve:Run the server the TUI connects to (started for you; internal)'
  )

  local -a specs
  specs=(
    '(--standalone --no-standalone)--standalone[Run with a private server instead of the background service]'
    '(--standalone --no-standalone)--no-standalone[Disable standalone]'
    '(--server)--server[Connect to a server URL instead of the background service]:value:'
    '(--auto --no-auto)--auto[Auto-approve permissions that are not explicitly denied]'
    '(--auto --no-auto)--no-auto[Disable auto]'
    '(--continue -c --no-continue)--continue[Continue the last session]'
    '(--continue -c --no-continue)-c[Continue the last session]'
    '(--continue -c --no-continue)--no-continue[Disable continue]'
    '(--session -s)--session[Session ID to continue, or to create if it does not exist]:value:'
    '(--session -s)-s[Session ID to continue, or to create if it does not exist]:value:'
    '(--prompt)--prompt[Prompt to use]:value:'
    -
    parent-arguments
    ':Directory to start Anysola Code in:'
    -
    subcommands
    '1:command:->command'
    '*::arg:->args'
  )

  _arguments -C "${specs[@]}"

  case "$state" in
    command)
      _describe -t commands 'commands' commands
      ;;
    args)
      case "$words[1]" in
        login)
          _anysola_login
          ;;
        logout)
          _anysola_logout
          ;;
        status)
          _anysola_status
          ;;
        usage)
          _anysola_usage
          ;;
        devices)
          _anysola_devices
          ;;
        update)
          _anysola_update
          ;;
        upgrade)
          _anysola_upgrade
          ;;
        uninstall)
          _anysola_uninstall
          ;;
        acp)
          _anysola_acp
          ;;
        debug)
          _anysola_debug
          ;;
        mcp)
          _anysola_mcp
          ;;
        plugin)
          _anysola_plugin
          ;;
        models)
          _anysola_models
          ;;
        run)
          _anysola_run
          ;;
        session)
          _anysola_session
          ;;
        service)
          _anysola_service
          ;;
        reload)
          _anysola_reload
          ;;
        serve)
          _anysola_serve
          ;;
      esac
      ;;
  esac
}
